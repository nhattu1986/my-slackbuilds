cmake_minimum_required(VERSION 3.24)

set(CMAKE_CXX_STANDARD 17)
find_package(Libinput REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PC_LIBUDEV libudev)

find_package(FLTK REQUIRED)
if(FLTK_FOUND AND NOT TARGET fltk::fltk)
  add_library(fltk::fltk INTERFACE IMPORTED)
  set_target_properties(
    fltk::fltk
    PROPERTIES
      INTERFACE_INCLUDE_DIRECTORIES "${FLTK_INCLUDE_DIR}"
      INTERFACE_LINK_LIBRARIES "${FLTK_LIBRARIES}"
  )
endif()
find_package(X11 REQUIRED)

add_executable(fcitx5_uinput_server fcitx5_uinput_server.cpp)
target_link_libraries(fcitx5_uinput_server PRIVATE Threads::Threads Libinput::Libinput -ludev) 

add_executable(sconfig main6.cpp)
target_link_libraries(sconfig PRIVATE Threads::Threads X11::X11 X11::Xext fltk::fltk)


install(TARGETS fcitx5_uinput_server DESTINATION "${CMAKE_INSTALL_PREFIX}/bin/")
install(TARGETS sconfig DESTINATION "${CMAKE_INSTALL_PREFIX}/bin/")
